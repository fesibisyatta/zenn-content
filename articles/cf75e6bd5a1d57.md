## 演習問題11.1

### $\boldsymbol{\beta_1}, \cdots, \boldsymbol{\beta_m}$ の完全条件付き分布

$$
\begin{aligned}
p(\boldsymbol{\beta}_j \mid \boldsymbol{y}_j, \boldsymbol{X}_j, \boldsymbol{\theta}, \boldsymbol{\Sigma}, \sigma^2) 

    &\propto p(\boldsymbol{y}_j \mid \boldsymbol{\beta}_j, \boldsymbol{X}_j, \boldsymbol{\theta}, \boldsymbol{\Sigma}, \sigma^2)\, p(\boldsymbol{\beta}_j, \boldsymbol{X}_j, \boldsymbol{\theta}, \boldsymbol{\Sigma}, \sigma^2) \\
 
    &\propto p(\boldsymbol{y}_j \mid \boldsymbol{\beta}_j, \boldsymbol{X}_j, \boldsymbol{\theta}, \boldsymbol{\Sigma}, \sigma^2)\, p(\boldsymbol{\beta}_j \mid  \boldsymbol{\theta}, \boldsymbol{\Sigma}) \\

    &\propto \exp \left(
        - \frac{1}{2\sigma^2} (\boldsymbol{y}_j - \boldsymbol{X}_j \boldsymbol{\beta}_j)^\top (\boldsymbol{y}_j - \boldsymbol{X}_j \boldsymbol{\beta}_j)
     \right) \\

    &\quad \cdot \exp \left(
        - \frac{1}{2} \boldsymbol{\beta}_j^\top \Sigma^{-1} \boldsymbol{\beta}_j + \boldsymbol{\beta}_j^\top \Sigma^{-1} \boldsymbol{\theta} 
    \right) \\

    &\propto \exp \left(
        -\frac{1}{2} \boldsymbol{\beta}_j^\top \left(\frac{ \boldsymbol{X_j}^\top \boldsymbol{X_j}}{\sigma^2} \right) \boldsymbol{\beta}_j + \boldsymbol{\beta}_j^\top \left(\frac{\boldsymbol{X}_j^\top \boldsymbol{y}_j}{\sigma^2} \right)
     \right) \\

    &\quad \cdot \exp \left(
        - \frac{1}{2} \boldsymbol{\beta}_j^\top \Sigma^{-1} \boldsymbol{\beta}_j + \boldsymbol{\beta}_j^\top \Sigma^{-1} \boldsymbol{\theta} 
    \right) \\

    &\propto \exp \left(
        - \frac{1}{2} \boldsymbol{\beta}_j^\top \Lambda^{-1} \boldsymbol{\beta}_j + \boldsymbol{\beta}_j^\top \left(\boldsymbol{\theta}^\top \boldsymbol{\Sigma}^{-1} + \frac{1}{\sigma^2}  \boldsymbol{X}_j^\top \boldsymbol{y}_j \right)  
    \right)

\end{aligned}
$$

となるので, $\displaystyle \Lambda = \left(\Sigma^{-1} + \frac{1}{\sigma^2} \boldsymbol{X}_j^\top \boldsymbol{X}_j \right)^{-1},\, \boldsymbol{\mu} = \Lambda \left(\boldsymbol{\theta}^\top \boldsymbol{\Sigma}^{-1} + \frac{1}{\sigma^2}  \boldsymbol{X}_j^\top \boldsymbol{y}_j \right)$ とおくと

$$
\boldsymbol{\beta}_j \mid \boldsymbol{y}_j, \boldsymbol{X}_j, \boldsymbol{\theta}, \boldsymbol{\Sigma}, \sigma^2 \sim \mathcal{N}\left(\mu, \Lambda \right)
$$

である. 

### $\boldsymbol{\theta}$ の完全条件付き分布

$$
\begin{aligned}
p(\boldsymbol{\theta} \mid \boldsymbol{\beta_1}, \cdots, \boldsymbol{\beta_m}, \boldsymbol{\Sigma}) 
    
    &\propto p(\boldsymbol{\beta_1}, \cdots, \boldsymbol{\beta_m}  \mid  \boldsymbol{\theta} , \boldsymbol{\Sigma})\, p(\boldsymbol{\theta}) \\
    
    &= \left(\prod_{j = 1}^m p(\boldsymbol{\beta_j} \mid \boldsymbol{\theta} , \boldsymbol{\Sigma}) \right) p(\boldsymbol{\theta}) \\
    
    &\propto \left(\prod_{j = 1}^m \exp \left(- \frac{1}{2} (\boldsymbol{\beta_j} - \boldsymbol{\theta})^\top \Sigma^{-1} (\boldsymbol{\beta_j} - \boldsymbol{\theta}) \right) \right) \\
    
    &\quad \cdot \exp \left(- \frac{1}{2} (\boldsymbol{\theta} - \boldsymbol{\mu_0})^\top \Lambda_0^{-1} (\boldsymbol{\theta} - \boldsymbol{\mu_0}) \right) \\
    
    &\propto \exp \left(- \frac{1}{2} \sum_{j = 1}^m (\boldsymbol{\beta_j} - \boldsymbol{\theta})^\top \Sigma^{-1} (\boldsymbol{\beta_j} - \boldsymbol{\theta}) \right) \\
    
    &\quad \cdot \exp \left(- \frac{1}{2} (\boldsymbol{\theta} - \boldsymbol{\mu_0})^\top \Lambda_0^{-1} (\boldsymbol{\theta} - \boldsymbol{\mu_0}) \right) \\
    
    &= \exp \left(- \frac{1}{2} \left(\boldsymbol{\theta}^\top \, (\Lambda_0^{-1} + m \Sigma^{-1})\, \boldsymbol{\theta} \right) + \boldsymbol{\theta}^\top (m \Sigma^{-1} \bar{\boldsymbol{\beta}} + \Lambda^{-1} \boldsymbol{\mu_0}) \right) \\
    
    &= \exp \left(- \frac{1}{2} \boldsymbol{\theta}^\top \, \Lambda_m^{-1} \, \boldsymbol{\theta} + \boldsymbol{\theta}^\top \Lambda_m^{-1} \boldsymbol{\mu_m} \right)

\end{aligned}
$$

従って

$$
\boldsymbol{\theta} \mid \boldsymbol{\beta_1}, \cdots, \boldsymbol{\beta_m}, \boldsymbol{\Sigma} \sim \mathcal{N}(\boldsymbol{\mu_m},\,  \boldsymbol{\Lambda_m})
$$

となる. ただし

$$\boldsymbol{\Lambda_m} := (\boldsymbol{\Lambda_0}^{-1} + m \boldsymbol{\Sigma}^{-1})^{-1},\quad \boldsymbol{\mu_m} := \boldsymbol{\Lambda_m} (m \boldsymbol{\Sigma}^{-1} \bar{\boldsymbol{\beta}} + \boldsymbol{\Lambda}^{-1} \boldsymbol{\mu_0})$$

である.

### $\boldsymbol{\Sigma}$ の完全条件付き分布

$$
\begin{aligned}

p(\boldsymbol{\Sigma} \mid \boldsymbol{\beta_1}, \cdots, \boldsymbol{\beta_m}, \boldsymbol{\theta}) 

    &\propto p(\boldsymbol{\beta_1}, \cdots, \boldsymbol{\beta_m} \mid \boldsymbol{\theta}, \boldsymbol{\Sigma})p(\boldsymbol{\Sigma}) \\

    &\propto |\boldsymbol{\Sigma}|^{-m/2} \exp \left(- \frac{1}{2} \sum_{j = 1}^m (\boldsymbol{\beta_j} - \boldsymbol{\theta})^\top \Sigma^{-1} (\boldsymbol{\beta_j} - \boldsymbol{\theta}) \right) \\

    &\quad \cdot |\boldsymbol{\Sigma}|^{ - \frac{1}{2}(\eta_0 + p + 1)} \exp \left(- \frac{1}{2} \operatorname{tr(\boldsymbol{S_0} \boldsymbol{\Sigma}^{-1}) } \right) \\

    &= |\boldsymbol{\Sigma}|^{ - \frac{1}{2}(m + \eta_0 + p + 1)} \exp \left( -\frac{1}{2} \operatorname{tr}(\boldsymbol{S_\theta} \boldsymbol{\Sigma}^{-1}) -\frac{1}{2} \operatorname{tr}(\boldsymbol{S_0} \boldsymbol{\Sigma}^{-1}) \right) \\

    &= |\boldsymbol{\Sigma}|^{ - \frac{1}{2}(m + \eta_0 + p + 1)} \exp \left( -\frac{1}{2} \operatorname{tr}((\boldsymbol{S_\theta}+ \boldsymbol{S_0}) \boldsymbol{\Sigma}^{-1}) \right) \\

\end{aligned}

$$

ただし,  $\displaystyle \boldsymbol{S_\theta} = \sum_{i = 1}^n (\boldsymbol{y}_i - \boldsymbol{\theta})^\top (\boldsymbol{y}_i - \boldsymbol{\theta})$ とおくことで, p125にある

$$
\sum_{i = 1}^n (\boldsymbol{y}_i - \boldsymbol{\theta})^\top \boldsymbol{\Sigma}^{-1} (\boldsymbol{y}_i - \boldsymbol{\theta}) = \operatorname{tr}(\boldsymbol{S_\theta} \boldsymbol{\Sigma}^{-1})
$$

を用いた. 従って

$$
\boldsymbol{\Sigma} \mid \boldsymbol{\beta_1}, \cdots, \boldsymbol{\beta_m}, \boldsymbol{\theta} \sim \operatorname{inverse-Wishert}(m + \eta_0, \boldsymbol{S_0} + \boldsymbol{S_\theta})
$$

となる.

### $\boldsymbol{\sigma^2}$ の完全条件付き分布

$$
\begin{aligned}
p(\sigma^2 \mid \boldsymbol{y_1}, \cdots, \boldsymbol{y_m})

    &\propto p(\boldsymbol{y_1}, \cdots, \boldsymbol{y_m} \mid \sigma^2)\, p(\sigma^2) \\

    &\propto \left(\prod_{j = 1}^m \prod_{i = 1}^{n_j} \frac{1}{\sqrt{2\pi \sigma^2}} \exp \left(-\frac{1}{2\sigma^2}(y_{ij} - \boldsymbol{\beta_j}^\top \boldsymbol{x}_{ij})^2 \right) \right) \\
    &\quad \cdot (\sigma^2)^{-\frac{\nu_0}{2} - 1} \exp \left(- \frac{\nu_0 \sigma_0^2}{2\sigma^2} \right) \\

    &\propto (\sigma^2)^{- \frac{\nu_0 + \sum_{j = 1}^m n_j}{2} - 1} \exp \left(-\frac{1}{2\sigma^2} \left(\nu_0 \sigma_0^2 + \sum_{j = 1}^m \sum_{i = 1}^{n_j}(y_{ij} - \boldsymbol{\beta_j}^\top \boldsymbol{x}_{ij})^2 \right) \right)


\end{aligned}
$$

となる. 従って

$$
SSR = \sum_{j = 1}^m \sum_{i = 1}^{n_j}(y_{ij} - \boldsymbol{\beta_j}^\top \boldsymbol{x}_{ij})^2
$$

とおくとき

$$
\sigma^2 \mid \boldsymbol{y_1}, \cdots, \boldsymbol{y_m} \sim \operatorname{inverse-gamma} \left(\frac{\nu_0 + \sum_{j = 1}^m n_j}{2},\, \frac{\nu_0 \sigma_0^2 + SSR}{2} \right)
$$

となる.
